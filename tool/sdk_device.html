1//! Device identity for query signing, login fingerprints, and `register_dev`.
2//!
3//! Host apps inject stable hardware / install identifiers via
4//! [`DeviceIdentity::builder`]; the SDK derives `mid` / `guid` and never
5//! silently re-rolls them. Login state lives in [`crate::Session`].

8use serde::{Deserialize, Serialize};

12/// How Standard-profile login fills body `t1` / `t2`.
13///
14/// Official apps generate t1/t2 in **`libj.so`** (`NativeParams` / `cc::d`/`cc::e`,
15/// AES-256-CBC). Open clients (KuGouMusicApi) send `t1=0` / `t2=0` / fixed `t3`
16/// on Standard. The experimental
17/// [`LiteAesOnStandard`](FingerprintMode::LiteAesOnStandard) reuses the same
18/// Lite AES scheme (now reverse-aligned with `libj`) on Standard — **opt-in only**.
19#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, Serialize, Deserialize)]
20#[serde(rename_all = "snake_case")]
21pub enum FingerprintMode {
22    /// Open-compatible: Standard → `t1=0`/`t2=0`/`t3=fixed`; Lite → AES t1/t2.
23    #[default]
24    OpenCompatible,
25    /// Emit Lite-style AES t1/t2 on Standard (same algorithm as official `libj`).
26    LiteAesOnStandard,
27}

29/// Stable device identifiers for query signing and login fingerprints.
30///
31/// # Production integration
32///
33/// Prefer [`DeviceIdentity::builder`] with a **stable** host device id
34/// (`android_id` / IMEI / app-install id) and persist the whole
35/// [`crate::Session`] (including `dfid` after `register_dev`).
36/// [`DeviceIdentity::random`] / [`crate::Session::random`] are for probes only —
37/// re-rolling mid/uuid every process looks like a new device and raises 风控.
38///
39/// # Official mid algorithm
40///
41/// Official Android: `mid = BigInteger(MD5(deviceId), 16).toString()` (decimal
42/// digit string of the MD5 digest). This is **not** the 32-char MD5 hex.
43/// See [`DeviceIdentity::mid_from_device_id`].
44///
45/// # Wire `uuid`
46///
47/// Official clients always send a **persisted** uuid (typically MD5 of
48/// `ANDROID_ID`) in the common query. Empty uuid falls back to `guid`, then
49/// `"-"`.
50#[derive(Clone, Serialize, Deserialize)]
51pub struct DeviceIdentity {
52    /// Device GUID (often 32-char MD5 hex of a stable device id).
53    ///
54    /// Used by Lite login `t2` plaintext and as a stable local id. When built
55    /// via [`DeviceIdentity::builder`] + `device_id`, defaults to
56    /// `MD5_hex(device_id)`.
57    pub guid: String,
58    /// Machine id used in query signing / headers.
59    ///
60    /// Official: decimal bigint form of `MD5(deviceId)`. See
61    /// [`DeviceIdentity::mid_from_device_id`].
62    pub mid: String,
63    /// MAC-like string injected into Lite login `t2` (default privacy MAC
64    /// `02:00:00:00:00:00` when unknown).
65    pub mac: String,
66    /// Short device code; **legacy** fallback for body `dev` when `model` is unset.
67    ///
68    /// Official t2 segment d uses `Build.MODEL` (`wire_dev`), not this field.
69    /// Builder default: first 10 chars of `guid` (stable). `random()` still
70    /// generates a fresh code.
71    pub device_code: String,
72    /// Server-issued device fingerprint (`dfid`) from `/risk/v2/r_register_dev`
73    /// or cookies. `None` until registered; wire uses `"-"` when absent.
74    /// Prefer [`Self::set_dfid`] / [`Self::has_dfid`].
75    pub device_fingerprint_id: Option<String>,
76    /// Query `uuid` (official: persisted MD5 of ANDROID_ID). Empty → transport
77    /// falls back to `guid`, then `"-"`.
78    #[serde(default)]
79    pub uuid: String,
80    /// Build.BRAND — register_dev fingerprint / display.
81    #[serde(default, skip_serializing_if = "Option::is_none")]
82    pub brand: Option<String>,
83    /// Build.MODEL — Lite body `dev` **and** t2 plaintext segment d.
84    #[serde(default, skip_serializing_if = "Option::is_none")]
85    pub model: Option<String>,
86    /// Build.MANUFACTURER.
87    #[serde(default, skip_serializing_if = "Option::is_none")]
88    pub manufacturer: Option<String>,
89    /// Build.SERIAL / build serial for register_dev.
90    #[serde(default, skip_serializing_if = "Option::is_none")]
91    pub build_serial: Option<String>,
92    /// Telephony IMEI. Official t2 middle segment = `MD5(imei)` (`libj` `h5`→`f8`).
93    /// Also used in `register_dev` fingerprint. Prefer setting when the host has it.
94    #[serde(default, skip_serializing_if = "Option::is_none")]
95    pub imei: Option<String>,
96    /// Optional ANDROID_ID for register_dev fingerprint (and often as `device_id`).
97    #[serde(default, skip_serializing_if = "Option::is_none")]
98    pub android_id: Option<String>,
99    /// Lite body `gitversion` (7-char hex-ish). Official pads app git version.
100    #[serde(default, skip_serializing_if = "Option::is_none")]
101    pub gitversion: Option<String>,
102    /// Optional per-session User-Agent. When set, transport overrides the
103    /// process-wide default from [`crate::KugouClientBuilder::user_agent`].
104    /// Prefer the client builder for a host-wide UA; use this when one
105    /// process holds multiple device sessions with different UAs.
106    #[serde(default, skip_serializing_if = "Option::is_none")]
107    pub user_agent: Option<String>,
108    /// Host-supplied login body `t1` (capture / host native). Takes priority over AES.
109    #[serde(default, skip_serializing_if = "Option::is_none")]
110    pub t1_override: Option<String>,
111    /// Host-supplied login body `t2`.
112    #[serde(default, skip_serializing_if = "Option::is_none")]
113    pub t2_override: Option<String>,
114    /// Host-supplied login body `t3`.
115    #[serde(default, skip_serializing_if = "Option::is_none")]
116    pub t3_override: Option<String>,
117    /// Extra JSON fields merged into default `register_dev` fingerprint.
118    #[serde(default, skip_serializing_if = "Option::is_none")]
119    pub fingerprint_extra: Option<Value>,
120    /// Standard-profile t1/t2 strategy (default open-compatible).
121    #[serde(default)]
122    pub fingerprint_mode: FingerprintMode,
123}

144    /// Probe-only random identity. **Do not use for production login** — prefer
145    /// [`Self::builder`] + persisted [`crate::Session`].
146    pub fn random() -> Self {
147        // Historical path: guid = MD5(UUID string), mid = decimal(MD5(guid)).
148        // uuid gets a stable 32-hex so wire is not always "-".

186    /// Set or clear server-issued `dfid`.
187    ///
188    /// Empty / `"-"` / `"null"` clears the field (wire will send `"-"`).
189    pub fn set_dfid(&mut self, dfid: impl Into<String>) {
190        let s = dfid.into();
191        let t = s.trim();
192        if t.is_empty() || t == "-" || t.eq_ignore_ascii_case("null") {
193            self.device_fingerprint_id = None;
194        } else {
195            self.device_fingerprint_id = Some(t.to_owned());
196        }
197    }
198
199    /// Whether a real server-issued `dfid` is present (not empty / `"-"`).
200    ///
201    /// SMS login and high-bitrate play often need this; call
202    /// [`crate::auth::AuthApi::register_dev`] when `false`.
203    pub fn has_dfid(&self) -> bool {
204        self.device_fingerprint_id
205            .as_deref()
206            .map(str::trim)
207            .is_some_and(|s| !s.is_empty() && s != "-" && !s.eq_ignore_ascii_case("null"))
208    }
209
210    /// Current `dfid` for diagnostics, or `"-"` when missing (same as wire).
211    pub fn dfid_or_dash(&self) -> &str {
212        self.device_fingerprint_id
213            .as_deref()
214            .map(str::trim)
215            .filter(|s| !s.is_empty() && *s != "-" && !s.eq_ignore_ascii_case("null"))
216            .unwrap_or("-")
217    }
218
219    /// Value for common query / header `uuid` (never empty for callers that
220    /// want a placeholder — returns `"-"` only when both uuid and guid empty).
221    pub fn wire_uuid(&self) -> &str {
222        let u = self.uuid.trim();
223        if !u.is_empty() && u != "-" {
224            return u;
225        }
226        let g = self.guid.trim();
227        if !g.is_empty() {
228            return g;
229        }
230        "-"
231    }
232
233    /// Lite body `dev`: prefer model, else device_code.
234    pub fn wire_dev(&self) -> &str {
235        self.model
236            .as_deref()
237            .map(str::trim)
238            .filter(|s| !s.is_empty())
239            .unwrap_or(self.device_code.as_str())
240    }
241
242    /// Lite `gitversion` or open-client placeholder.
243    pub fn wire_gitversion(&self) -> &str {
244        self.gitversion
245            .as_deref()
246            .map(str::trim)
247            .filter(|s| !s.is_empty())
248            .unwrap_or("5f0b7c4")
249    }

408    /// Build a complete [`DeviceIdentity`].
409    ///
410    /// Requires at least one of: `device_id`, `android_id`, `imei`, `mid`, or `guid`.
411    /// When `device_id` is omitted, `android_id` then `imei` are used as the stable
412    /// id source for mid/guid derivation (common host-app pattern).
413    ///
414    /// **mid-only:** `.mid("…")` alone is valid — `guid`/`uuid` default to the mid
415    /// string so query signing still has a stable identifier.
416    pub fn build(self) -> crate::Result<DeviceIdentity> {
417        // Prefer explicit device_id; fall back to android_id / imei so hosts can
418        // pass a single telephony/settings field without duplicating it.
419        let device_id_owned = self
420            .device_id
421            .as_deref()
422            .map(str::trim)
423            .filter(|s| !s.is_empty())
424            .map(|s| s.to_owned())
425            .or_else(|| {
426                self.android_id
427                    .as_deref()
428                    .map(str::trim)
429                    .filter(|s| !s.is_empty())
430                    .map(|s| s.to_owned())
431            })
432            .or_else(|| {
433                self.imei
434                    .as_deref()
435                    .map(str::trim)
436                    .filter(|s| !s.is_empty())
437                    .map(|s| s.to_owned())
438            });
439        let device_id = device_id_owned.as_deref();
440        let explicit_mid = self
441            .mid
442            .as_deref()
443            .map(str::trim)
444            .filter(|s| !s.is_empty())
445            .map(|s| s.to_owned());
446
447        let guid = if let Some(g) = self.guid.filter(|s| !s.trim().is_empty()) {
448            g.trim().to_owned()
449        } else if let Some(id) = device_id {
450            md5_hex(id.as_bytes())
451        } else if let Some(ref m) = explicit_mid {
452            // mid-only: reuse mid as the local stable id (host already has a
453            // server-side mid; no device_id to hash).
454            m.clone()
455        } else {
456            return Err(crate::KugouError::InvalidArgument(
457                "DeviceIdentityBuilder needs device_id, android_id, imei, guid, or mid".into(),
458            ));
459        };
460
461        let mid = if let Some(m) = explicit_mid {
462            m
463        } else if let Some(id) = device_id {
464            // Official single-layer: mid = decimal(MD5(device_id))
465            DeviceIdentity::mid_from_device_id(id)
466        } else {
467            // guid-only path (legacy-compatible): mid = decimal(MD5(guid))
468            calculate_mid(&guid)
469        };
470
471        let uuid = if let Some(u) = self.uuid.filter(|s| !s.trim().is_empty()) {
472            u.trim().to_owned()
473        } else {
474            guid.clone()
475        };
476
477        let mac = self
478            .mac
479            .filter(|s| !s.trim().is_empty())
480            .map(|s| s.trim().to_owned())
481            .unwrap_or_else(|| "02:00:00:00:00:00".into());
482
483        // Prefer host device_code; else first 10 of guid (stable across rebuilds of
484        // the same identity). Only fall back to random when guid is empty.
485        let device_code = self
486            .device_code
487            .filter(|s| !s.trim().is_empty())
488            .map(|s| s.trim().to_owned())
489            .unwrap_or_else(|| {
490                let g = guid.trim();
491                if g.is_empty() {
492                    Uuid::new_v4().simple().to_string()[..10].to_ascii_uppercase()
493                } else {
494                    let mut code = g.to_ascii_uppercase();
495                    code.truncate(10);
496                    while code.len() < 10 {
497                        code.push('0');
498                    }
499                    code
500                }
501            });
502
503        // Trim optional string fields so host " MI 14 " does not leak spaces into wire.
504        let opt_trim = |o: Option<String>| o.map(|s| s.trim().to_owned()).filter(|s| !s.is_empty());
505
506        let mut device = DeviceIdentity {
507            guid,
508            mid,
509            mac,
510            device_code,
511            device_fingerprint_id: None,
512            uuid,
513            brand: opt_trim(self.brand),
514            model: opt_trim(self.model),
515            manufacturer: opt_trim(self.manufacturer),
516            build_serial: opt_trim(self.build_serial),
517            imei: opt_trim(self.imei),
518            android_id: opt_trim(self.android_id),
519            gitversion: opt_trim(self.gitversion),
520            user_agent: opt_trim(self.user_agent),
521            t1_override: opt_trim(self.t1_override),
522            t2_override: opt_trim(self.t2_override),
523            t3_override: opt_trim(self.t3_override),
524            fingerprint_extra: self.fingerprint_extra,
525            fingerprint_mode: self.fingerprint_mode,
526        };
527        if let Some(dfid) = self.dfid {
528            device.set_dfid(dfid);
529        }
530        Ok(device)
531    }
532}

540/// Decimal digit string of `MD5(value)` digest (big-endian), matching official
541/// `BigInteger(md5Hex, 16).toString()`.
542pub fn calculate_mid(value: &str) -> String {
543    let digest = Md5::digest(value.as_bytes());
544    digest.iter().fold(String::from("0"), |decimal, byte| {
545        mul_add_decimal(&decimal, 256, *byte as u32)
546    })
547}
548
549fn md5_hex(bytes: &[u8]) -> String {
550    format!("{:x}", Md5::digest(bytes))
551}
552
553fn mul_add_decimal(value: &str, multiplier: u32, addend: u32) -> String {
554    let mut carry = addend;
555    let mut out = Vec::with_capacity(value.len() + 3);
556    for digit in value.bytes().rev() {
557        let n = (digit - b'0') as u32 * multiplier + carry;
558        out.push((n % 10) as u8 + b'0');
559        carry = n / 10;
560    }
561    while carry > 0 {
562        out.push((carry % 10) as u8 + b'0');
563        carry /= 10;
564    }
565    out.reverse();
566    String::from_utf8(out).expect("decimal digits are UTF-8")
567}
