7/// Upstream anti-automation challenge. The SDK reports it but does not simulate a browser.
8///
9/// Raised when response headers include `ssa-code` (or similar). Callers that need
10/// to pass SSA must solve the challenge out-of-band; this crate only surfaces it.
11#[derive(Debug, Clone, Serialize, Deserialize)]
12#[non_exhaustive]
13pub struct SecurityChallenge {
14    /// Challenge code from the `ssa-code` response header.
15    pub code: String,
16    /// Parsed response body (often includes challenge payload / tips).
17    pub body: Value,
18}

20/// Library error type for transport, protocol, auth, and decode failures.
21#[derive(Debug, Error)]
22#[non_exhaustive]
23pub enum KugouError {
24    /// Caller-supplied argument failed validation (empty key, bad mobile, …).
25    #[error("invalid argument: {0}")]
26    InvalidArgument(String),
27    /// API requires an authenticated [`crate::Session`] (userid + token).
28    #[error("authentication is required for this operation")]
29    AuthenticationRequired,
30    /// Underlying HTTP client error (connect / timeout / TLS / …).
31    ///
32    /// The wrapped [`reqwest::Error`] has its related URL stripped via
33    /// [`reqwest::Error::without_url`] — KuGou query strings often carry
34    /// `token` / `signature`, and reqwest's `Display` would otherwise include them.
35    #[error("HTTP transport failed: {0}")]
36    Transport(reqwest::Error),
37    /// JSON serialize/deserialize failure.
38    #[error("serialization failed: {0}")]
39    Serialization(#[from] serde_json::Error),
40    /// Non-success HTTP status from KuGou (or gateway).
41    #[error("KuGou endpoint {endpoint} returned HTTP {status}: {body}")]
42    Http {
43        /// Service path used by the failed request.
44        endpoint: String,
45        /// HTTP status code.
46        status: u16,
47        /// Truncated response body text for diagnostics.
48        body: String,
49    },
50    /// A successful HTTP response could not be interpreted as the expected protocol.
51    #[error("invalid response from {endpoint}: {message}")]
52    Protocol {
53        /// Service path used by the failed request.
54        endpoint: String,
55        /// Content type returned by the server, when available.
56        content_type: Option<String>,
57        /// Human-readable parse or integrity problem.
58        message: String,
59        /// Truncated response body text for diagnostics.
60        body: String,
61    },
62    /// Upstream JSON envelope reported failure (`status=0` and/or non-zero `error_code`).
63    #[error("KuGou business error {code}: {message}")]
64    Business {
65        /// Prefer `error_code` when non-zero; else envelope `status`.
66        code: i64,
67        /// Best-effort message from `error_msg` / `errmsg` / `message` / `msg`.
68        message: String,
69        /// Full parsed JSON body for diagnostics.
70        body: Value,
71    },
72    /// SSA / anti-bot challenge — see [`SecurityChallenge`].
73    #[error("security verification required: {0:?}")]
74    SecurityChallenge(SecurityChallenge),
75    /// AES/RSA/KRC or other binary decode failure.
76    #[error("cryptographic or lyric decoding failed: {0}")]
77    Decode(String),
78    /// Client builder / clock / config problem.
79    #[error("client configuration error: {0}")]
80    Configuration(String),
81    /// Envelope looked successful but the soft body is missing required fields
82    /// for a typed API (e.g. create playlist without a new `list_id`).
83    ///
84    /// Raw `*_raw` methods do not raise this — they return the JSON for inspection.
85    #[error("unexpected response from {endpoint}: {issue}")]
86    UnexpectedResponse {
87        /// Short endpoint label (e.g. `playlist.create`).
88        endpoint: &'static str,
89        /// Human-readable integrity failure.
90        issue: String,
91        /// Soft-parsed or raw business payload for diagnostics.
92        data: Value,
93    },
94}
