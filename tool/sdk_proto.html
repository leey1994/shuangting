14pub(crate) struct RequestSpec {
15    pub method: Method,
16    pub service: Service,
17    pub path: String,
18    pub query: BTreeMap<String, Value>,
19    /// JSON body (mutually exclusive with [`Self::raw_body`] in practice).
20    pub body: Option<Value>,
21    /// Raw request body + Content-Type (e.g. AES Base64 ciphertext for `register_dev`).
22    ///
23    /// When set, these bytes are used for Android signature material and the wire body.
24    pub raw_body: Option<(Vec<u8>, &'static str)>,
25    pub router: Option<&'static str>,
26    pub signature: SignatureKind,
27    pub extra_headers: Vec<(&'static str, String)>,
28}

30impl RequestSpec {
31    pub fn get(path: impl Into<String>) -> Self {
32        Self {
33            method: Method::GET,
34            service: Service::Gateway,
35            path: path.into(),
36            query: BTreeMap::new(),
37            body: None,
38            raw_body: None,
39            router: None,
40            signature: SignatureKind::Android,
41            extra_headers: Vec::new(),
42        }
43    }
44    pub fn post(path: impl Into<String>) -> Self {
45        Self {
46            method: Method::POST,
47            ..Self::get(path)
48        }
49    }
50    pub fn service(mut self, value: Service) -> Self {
51        self.service = value;
52        self
53    }
54    pub fn query<T: serde::Serialize>(mut self, value: &T) -> Result<Self> {
55        self.query.extend(to_map(value)?);
56        Ok(self)
57    }
58    pub fn body<T: serde::Serialize>(mut self, value: &T) -> Result<Self> {
59        self.body = Some(serde_json::to_value(value)?);
60        Ok(self)
61    }
62    /// Raw body bytes with an explicit Content-Type (not JSON-serialized).
63    pub fn raw_body(mut self, bytes: impl Into<Vec<u8>>, content_type: &'static str) -> Self {
64        self.raw_body = Some((bytes.into(), content_type));
65        self
66    }
67    pub fn router(mut self, value: &'static str) -> Self {
68        self.router = Some(value);
69        self
70    }
71    pub fn signature(mut self, value: SignatureKind) -> Self {
72        self.signature = value;
73        self
74    }
75    pub fn header(mut self, name: &'static str, value: impl Into<String>) -> Self {
76        self.extra_headers.push((name, value.into()));
77        self
78    }

89pub(crate) async fn execute<T: DeserializeOwned>(
90    client: &KugouClient,
91    session: &mut Session,
92    spec: RequestSpec,
93) -> Result<ApiResponse<T>> {
94    let tr = transport(client, session, spec).await?;
95    let root: Value = serde_json::from_slice(&tr.bytes).map_err(|e| KugouError::Protocol {
96        endpoint: tr.endpoint.clone(),
97        content_type: tr.content_type.clone(),
98        message: format!("response body is not valid JSON: {e}"),
99        body: diagnostic_body(&tr.bytes, false),
100    })?;
101    parse_envelope(session, tr.endpoint, &tr.headers, tr.content_type, root)
102}

116/// Parse a KuGou JSON envelope into [`ApiResponse`], applying session side-effects.
117pub(crate) fn parse_envelope<T: DeserializeOwned>(
118    session: &mut Session,
119    _endpoint: String,
120    headers: &HeaderMap,
121    _content_type: Option<String>,
122    root: Value,
123) -> Result<ApiResponse<T>> {
124    let upstream_status = envelope_i64(&root, "status");
125    let error_code = envelope_i64(&root, "error_code").unwrap_or_default();
126    if upstream_status == Some(0) || error_code != 0 {
127        let message = root
128            .get("error_msg")
129            .or_else(|| root.get("errmsg"))
130            .or_else(|| root.get("message"))
131            .or_else(|| root.get("msg"))
132            .and_then(Value::as_str)
133            .unwrap_or("unknown upstream error")
134            .to_owned();
135        let code = if error_code != 0 {
136            error_code
137        } else {
138            upstream_status.unwrap_or_default()
139        };
140        // Business failures are expected Results for hosts — debug only, not warn.
141        log::debug!(
142            target: "kugou_sdk::protocol",
143            "business error code={code} message={message}"
144        );
145        return Err(KugouError::Business {
146            code,
147            message,
148            body: root,
149        });
150    }
151    apply_body_session(session, &root);
152    let data_value = root.get("data").cloned().unwrap_or_else(|| root.clone());
153    Ok(ApiResponse {
154        data: serde_json::from_value(data_value)?,
155        upstream_status,
156        headers: diagnostic_headers(headers),
157        extra: envelope_without_data(&root),
158    })
159}

161async fn transport(
162    client: &KugouClient,
163    session: &mut Session,
164    mut spec: RequestSpec,
165) -> Result<TransportResponse> {
166    let now = SystemTime::now()
167        .duration_since(UNIX_EPOCH)
168        .map_err(|e| KugouError::Configuration(e.to_string()))?
169        .as_secs();
170    let defaults = [
171        ("appid", json!(client.platform.app_id())),
172        ("clientver", json!(client.platform.client_version())),
173        ("clienttime", json!(now)),
174        ("dfid", json!(session.device.dfid_or_dash())),
175        ("mid", json!(session.device.mid)),
176        // Official ParamGenerator always sends a persisted uuid; "-" is a bot signal.
177        ("uuid", json!(session.device.wire_uuid())),
178    ];
179    for (k, v) in defaults {
180        spec.query.entry(k.into()).or_insert(v);
181    }
182    // Official clients always send token/userid (empty / 0 when anonymous).
183    spec.query
184        .entry("token".into())
185        .or_insert(json!(session.token.as_deref().unwrap_or("")));
186    spec.query
187        .entry("userid".into())
188        .or_insert(json!(session.user_id.unwrap_or(0).to_string()));
189
190    // Take body once: same bytes for signature, wire, and (when enabled) trace.
191    let (body_bytes, content_type_req) = match (spec.raw_body.take(), spec.body.take()) {
192        (Some((raw, ct)), _) => (raw, Some(ct)),
193        (None, Some(v)) => (serde_json::to_vec(&v)?, Some("application/json")),
194        (None, None) => (Vec::new(), None),
195    };
196    if let Some(sig) = signing::signature(spec.signature, client.platform, &spec.query, &body_bytes)
197    {
198        spec.query.insert("signature".into(), json!(sig));
199    }
200    let endpoint = format!("{:?} {}", spec.service, spec.path);
201    let url = client
202        .service_url(spec.service)
203        .join(spec.path.trim_start_matches('/'))
204        .map_err(|e| KugouError::Configuration(format!("invalid endpoint {endpoint}: {e}")))?;
205    // Final URL with query — wire request + log lines share this Url.
206    let mut url_with_query = url;
207    {
208        let mut pairs = url_with_query.query_pairs_mut();
209        pairs.clear();
210        for (k, v) in &spec.query {
211            pairs.append_pair(k, &scalar_query_value(v));
212        }
213    }
214    // Keep Method for request + debug after send (`http::Method` is Clone, not Copy).
215    let method = spec.method;
216    let router = spec.router;
217    let extra_headers = std::mem::take(&mut spec.extra_headers);
218    let session_ua = session
219        .device
220        .user_agent
221        .as_deref()
222        .map(str::trim)
223        .filter(|s| !s.is_empty());
224    let dfid = session.device.dfid_or_dash();
225    let mid = session.device.mid.as_str();
226    let req_len = body_bytes.len();
227
228    // Outer gate: `log!` only checks global max_level; target filters run later.
229    // Multi-KB body pretty-print must not run when only other crates are at trace.
230    if log::log_enabled!(target: "kugou_sdk::protocol", log::Level::Trace) {
231        log::trace!(
232            target: "kugou_sdk::protocol",
233            "HTTP request\n{}",
234            format_http_request_trace(
235                method.as_str(),
236                url_with_query.as_str(),
237                dfid,
238                mid,
239                now,
240                session_ua,
241                router,
242                &extra_headers,
243                content_type_req,
244                &body_bytes,
245            )
246        );
247    }
248
249    // reqwest takes Method by value (not Into); Clone is cheap for standard verbs.
250    let mut request = client
251        .http
252        .request(method.clone(), url_with_query.as_str())
253        .header("dfid", dfid)
254        .header("mid", mid)
255        .header("clienttime", now)
256        .header("kg-rc", "1")
257        .header("kg-thash", "5d816a0")
258        .header("kg-rec", "1")
259        .header("kg-rf", "B9EDA08A64250DEFFBCADDEE00F8F25F");
260    // Session-level UA overrides the client-default User-Agent when host set one
261    // on DeviceIdentity (e.g. real-device string). Prefer KugouClientBuilder::user_agent
262    // for a process-wide default; this is per-session.
263    if let Some(ua) = session_ua {
264        request = request.header(reqwest::header::USER_AGENT, ua);
265    }
266    if let Some(router) = router {
267        request = request.header("x-router", router);
268    }
269    for (name, value) in &extra_headers {
270        request = request.header(*name, value);
271    }
272    if let Some(ct) = content_type_req {
273        request = request.header(reqwest::header::CONTENT_TYPE, ct);
274    }
275    if !body_bytes.is_empty() {
276        request = request.body(body_bytes);
277    }
278
279    let started = Instant::now();
280    let response = match request.send().await {
281        Ok(r) => r,
282        Err(e) => {
283            // `From` strips related URL (reqwest Display would embed token/signature).
284            let err = KugouError::from(e);
285            log::warn!(
286                target: "kugou_sdk::protocol",
287                "HTTP send failed endpoint={endpoint} err={err}"
288            );
289            return Err(err);
290        }
291    };
292    let status = response.status();
293    let headers = response.headers().clone();
294    update_session(session, &headers);
295    let challenge_code = headers
296        .get("ssa-code")
297        .and_then(|v| v.to_str().ok())
298        .map(str::to_owned);
299    let content_type = headers
300        .get(reqwest::header::CONTENT_TYPE)
301        .and_then(|v| v.to_str().ok())
302        .map(str::to_owned);
303    let limit = if !status.is_success() || challenge_code.is_some() {
304        ERROR_BODY_LIMIT
305    } else {
306        SUCCESS_BODY_LIMIT
307    };
308    let (bytes, truncated) = read_body_limited(response, limit).await?;
309    let elapsed = started.elapsed();
310
311    if log::log_enabled!(target: "kugou_sdk::protocol", log::Level::Trace) {
312        log::trace!(
313            target: "kugou_sdk::protocol",
314            "HTTP response\n{}",
315            format_http_response_trace(status.as_u16(), &headers, &bytes, truncated)
316        );
317    }
318    // redact_url_query is cheap; rely on log! max_level gate.
319    log::debug!(
320        target: "kugou_sdk::protocol",
321        "{method} {url} → {status} elapsed={elapsed_ms}ms req_len={req_len} resp_len={resp_len}{trunc}",
322        url = redact_url_query(url_with_query.as_str()),
323        status = status.as_u16(),
324        elapsed_ms = elapsed.as_millis(),
325        resp_len = bytes.len(),
326        trunc = if truncated { " truncated" } else { "" },
327    );
328
329    if let Some(code) = challenge_code {
330        log::warn!(
331            target: "kugou_sdk::protocol",
332            "security challenge ssa-code={code} endpoint={endpoint}"
333        );
334        let body_text = diagnostic_body(&bytes, truncated);
335        let body = serde_json::from_slice(&bytes).unwrap_or(Value::String(body_text));
336        return Err(KugouError::SecurityChallenge(SecurityChallenge {
337            code,
338            body,
339        }));
340    }
341    if !status.is_success() {
342        log::warn!(
343            target: "kugou_sdk::protocol",
344            "HTTP {status} endpoint={endpoint} body_len={}",
345            bytes.len(),
346            status = status.as_u16(),
347        );
348        return Err(KugouError::Http {
349            endpoint,
350            status: status.as_u16(),
351            body: diagnostic_body(&bytes, truncated),
352        });
353    }
354    if truncated {
355        log::warn!(
356            target: "kugou_sdk::protocol",
357            "response body truncated at {SUCCESS_BODY_LIMIT} bytes endpoint={endpoint}"
358        );
359        return Err(KugouError::Protocol {
360            endpoint,
361            content_type,
362            message: format!("response body exceeds {SUCCESS_BODY_LIMIT} bytes"),
363            body: diagnostic_body(&bytes, truncated),
364        });
365    }
366    Ok(TransportResponse {
367        endpoint,
368        headers,
369        content_type,
370        bytes,
371    })
372}

374/// Query value as sent on the wire (strings unquoted; numbers/bools as JSON tokens).
375fn scalar_query_value(value: &Value) -> String {
376    match value {
377        Value::String(s) => s.clone(),
378        Value::Null => String::new(),
379        other => other.to_string(),
380    }

386async fn read_body_limited(
387    mut response: reqwest::Response,
388    limit: usize,
389) -> Result<(Vec<u8>, bool)> {
390    let mut body = Vec::new();
391    while let Some(chunk) = response.chunk().await? {
392        let remaining = limit.saturating_sub(body.len());
393        if chunk.len() > remaining {
394            body.extend_from_slice(&chunk[..remaining]);
395            return Ok((body, true));
396        }
397        body.extend_from_slice(&chunk);
398    }
399    Ok((body, false))
400}

402fn diagnostic_body(bytes: &[u8], truncated: bool) -> String {
403    let mut body = String::from_utf8_lossy(bytes).into_owned();
404    if truncated {
405        body.push_str("…<truncated>");
406    }
407    body
408}

410fn envelope_i64(root: &Value, key: &str) -> Option<i64> {
411    let value = root.get(key)?;
412    value
413        .as_i64()
414        .or_else(|| value.as_u64().and_then(|n| i64::try_from(n).ok()))
415        .or_else(|| value.as_str().and_then(|s| s.trim().parse().ok()))
416}

418fn to_map<T: serde::Serialize>(value: &T) -> Result<BTreeMap<String, Value>> {

427fn update_session(session: &mut Session, headers: &HeaderMap) {
428    for value in headers.get_all(reqwest::header::SET_COOKIE) {
429        if let Ok(raw) = value.to_str()
430            && let Some((name, rest)) = raw.split_once('=')
431        {
432            session.apply_cookie(name.trim(), rest.split(';').next().unwrap_or_default());
433        }
434    }
435}

437fn apply_body_session(session: &mut Session, root: &Value) {
438    if let Some(data) = root.get("data").and_then(Value::as_object) {
439        for name in ["token", "userid", "vip_token", "vip_type", "dfid"] {
440            if let Some(v) = data.get(name) {
441                session.apply_json_field(name, v);
442            }
443        }
444    }
445}

447/// Protocol envelope fields only — never a copy of the business `data` payload.
448///
449/// When the upstream body has no `data` key (root-body endpoints such as lyrics),
450/// `ApiResponse.data` already holds the business body, so `extra` is empty.
451fn envelope_without_data(root: &Value) -> Value {
452    let Some(obj) = root.as_object() else {
453        return Value::Object(Default::default());
454    };
455    if !obj.contains_key("data") {
456        // Root-body response: the whole object is the business payload (also in `data`).
457        return Value::Object(Default::default());
458    }
459    let mut envelope = serde_json::Map::new();
460    for (k, v) in obj {
461        if k != "data" {
462            envelope.insert(k.clone(), v.clone());
463        }
464    }
465    Value::Object(envelope)
466}

468/// Safe diagnostic headers only — never `Set-Cookie` or authorization material.
469fn diagnostic_headers(headers: &HeaderMap) -> BTreeMap<String, String> {
470    const ALLOW: &[&str] = &[
471        "content-type",
472        "content-length",
473        "ssa-code",
474        "x-request-id",
475        "x-kg-request-id",
476        "kg-request-id",
477        "x-trace-id",
478        "traceparent",
479        "x-ratelimit-limit",
480        "x-ratelimit-remaining",
481        "x-ratelimit-reset",
482        "retry-after",
483    ];
484    headers
485        .iter()
486        .filter_map(|(k, v)| {
487            let name = k.as_str();
488            if !ALLOW.iter().any(|a| name.eq_ignore_ascii_case(a)) {
489                return None;
490            }
491            v.to_str().ok().map(|v| (name.to_owned(), v.to_owned()))
492        })
493        .collect()
494}
