1use crate::PlatformProfile;
2use md5::{Digest, Md5};
3use serde_json::Value;
4use std::collections::BTreeMap;

6pub(crate) enum SignatureKind {
7    Android,
8    Web,
9}
10
11pub(crate) fn signature(
12    kind: SignatureKind,
13    profile: PlatformProfile,
14    params: &BTreeMap<String, Value>,
15    body: &[u8],
16) -> Option<String> {
17    let pairs = params
18        .iter()
19        .map(|(k, v)| format!("{k}={}", scalar(v)))
20        .collect::<String>();
21    match kind {
22        SignatureKind::Android => Some(md5_parts(&[
23            profile.android_salt().as_bytes(),
24            pairs.as_bytes(),
25            body,
26            profile.android_salt().as_bytes(),
27        ])),
28        SignatureKind::Web => {
29            let salt = b"NVPh5oo715z5DIWAeQlhMDsWXXQV4hwt";
30            Some(md5_parts(&[salt, pairs.as_bytes(), salt]))
31        }
32    }
33}

55fn scalar(value: &Value) -> String {
56    match value {
57        Value::String(v) => v.clone(),
58        _ => value.to_string(),
59    }
60}
61pub(crate) fn md5_parts(parts: &[&[u8]]) -> String {
62    let mut h = Md5::new();
63    for p in parts {
64        h.update(p);
65    }
66    format!("{:x}", h.finalize())
67}
